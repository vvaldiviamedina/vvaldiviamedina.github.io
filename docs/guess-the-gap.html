Adivina la brecha

Un mini-juego sobre la brecha de género en el trabajo doméstico — ENUT Chile 2023

¿Qué tan bien calibrada tienes la intuición sobre la brecha de género en el trabajo doméstico no remunerado? Este juego te muestra cuatro grupos etarios, uno a la vez, y te pide adivinar cuánto tiempo más al día dedican las mujeres al trabajo doméstico, comparado con los hombres. Después de cada ronda te mostramos el valor real.

NoteFuente de los datos

Instituto Nacional de Estadísticas (INE) de Chile, II Encuesta Nacional sobre Uso del Tiempo (II ENUT) 2023, personas de 15 años y más. Los promedios de horas diarias de trabajo doméstico no remunerado están ponderados por el factor de expansión de personas (fe_cut) — los mismos datos usados en la página Trabajo pago y doméstico.

function shuffle(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}
function getHours(activity, ageTab, sex) {
  const rec = durationRaw.find(d => d.activity === activity && d.age_tab === ageTab && d.sex === sex);
  return rec ? rec.hours : null;
}
function roundData(ageTab) {
  const hoursM = getHours("Trabajo doméstico", ageTab, "Mujeres");
  const hoursH = getHours("Trabajo doméstico", ageTab, "Hombres");
  const gapMin = Math.round((hoursM - hoursH) * 60);
  return { ageTab, hoursM, hoursH, gapMin };
}
function feedback(errorAbs, actualGap) {
  const pct = errorAbs / actualGap;
  if (pct <= 0.1) return { emoji: "🎯", text: "¡Exacto! Tu intuición está muy bien calibrada." };
  if (pct <= 0.25) return { emoji: "👏", text: "Muy cerca." };
  if (pct <= 0.5) return { emoji: "🤔", text: "No tan mal, pero hay una brecha entre tu intuición y el dato." };
  return { emoji: "😮", text: "Bastante lejos — la brecha real sorprende." };
}
function formatMin(min) {
  const hh = Math.floor(min / 60);
  const mm = min % 60;
  if (hh === 0) return `${mm} min`;
  return mm === 0 ? `${hh} h` : `${hh} h ${mm} min`;
}

    const maxHours = Math.max(data.hoursM, data.hoursH) * 1.15;
    const barScale = h => (h / maxHours) * (svgWidth - 140);

      { label: "Mujeres", hours: data.hoursM, color: colorM, y: 10 },
      { label: "Hombres", hours: data.hoursH, color: colorH, y: 50 }

  function renderSummary() {
    roundBox.innerHTML = "";
    const card = html`<div style="max-width: 640px; margin: 0 auto; padding: 24px; border-radius: 10px; background: color-mix(in srgb, currentColor 5%, transparent);"></div>`;

    const title = html`<div style="font-weight: 700; font-size: 1.2rem; margin-bottom: 16px;">Resultado final</div>`;

      <th style="text-align:right; padding:6px; border-bottom: 1px solid #999;">Tu respuesta</th>
      <th style="text-align:right; padding:6px; border-bottom: 1px solid #999;">Real</th>
      <th style="text-align:right; padding:6px; border-bottom: 1px solid #999;">Diferencia</th>
    </tr></thead>`;
    table.appendChild(thead);
    const tbody = html`<tbody></tbody>`;
    let totalError = 0;
    results.forEach(r => {
      const err = Math.abs(r.guess - r.actual);
      totalError += err;
      const row = html`<tr>
        <td style="padding:6px; border-bottom: 1px solid #eee;">${ageTabLabel[r.ageTab]}</td>
        <td style="text-align:right; padding:6px; border-bottom: 1px solid #eee;">${formatMin(r.guess)}</td>
        <td style="text-align:right; padding:6px; border-bottom: 1px solid #eee;">${formatMin(r.actual)}</td>
        <td style="text-align:right; padding:6px; border-bottom: 1px solid #eee;">${formatMin(err)}</td>
      </tr>`;
      tbody.appendChild(row);
    });
    table.appendChild(tbody);

    const avgError = Math.round(totalError / results.length);
    const closing = html`<div style="margin-bottom: 20px;">Tu error promedio fue de <strong>${formatMin(avgError)}</strong> por ronda. ${avgError <= 15 ? "Muy buena intuición sobre la brecha de género en el trabajo doméstico." : avgError <= 35 ? "Una intuición razonable, aunque la brecha real suele ser más marcada de lo que se piensa." : "La brecha real es más grande de lo que la mayoría intuye — así es la desigualdad en el trabajo doméstico no remunerado."}</div>`;

    const replayBtn = html`<button style="padding:8px 20px;border-radius:4px;border:1px solid #5c9e95;background:#5c9e95;color:white;cursor:pointer;font-size:0.95rem;">Jugar de nuevo</button>`;
    replayBtn.onclick = () => {
      order = shuffle(ageTabOrder);
      roundIndex = 0;
      results = [];
      renderRound();
    };

    card.appendChild(title);
    card.appendChild(table);
    card.appendChild(closing);
    card.appendChild(replayBtn);
    roundBox.appendChild(card);
  }

  renderRound();
  return wrapper;
}
Important

El trabajo doméstico no remunerado es una de las brechas de género más marcadas y persistentes en el uso del tiempo: se mantiene en todos los grupos etarios, y no se explica solo por diferencias en la participación laboral remunerada — revisa la página Trabajo pago y doméstico para más detalle.

function buildGaugeDurationDemo(container) {
  const ageTab = randomAgeTab();
  const data = roundData(ageTab);
  const GMIN = 0, GMAX = 150;
  const R = 140, width = 2 * R + 100, height = R + 90;
  const cx = width / 2, cy = height - 55;

  function angleFor(v) {
    const t = Math.max(GMIN, Math.min(GMAX, v));
    return ((t - GMIN) / (GMAX - GMIN) - 0.5) * Math.PI;
  }
  function needlePoint(angle, len) {
    return [cx + Math.sin(angle) * len, cy - Math.cos(angle) * len];
  }

  const card = html`<div style="max-width: 640px; margin: 0 auto; padding: 24px; border-radius: 10px; background: color-mix(in srgb, currentColor 5%, transparent); text-align:center;"></div>`;
  const group = html`<div style="font-weight: 700; font-size: 1.1rem; margin-bottom: 8px;">${ageTabLabel[ageTab]}</div>`;
  const question = html`<div style="margin-bottom: 12px; max-width: 480px; margin-left:auto; margin-right:auto;">¿Cuánto <strong>tiempo más al día</strong> dedican las <span style="color:${colorM}; font-weight:700;">mujeres</span> al trabajo doméstico, comparado con los <span style="color:${colorH}; font-weight:700;">hombres</span>? Mueve la aguja.</div>`;

  const gaugeHolder = html`<div></div>`;
  const svg = d3.select(gaugeHolder).append("svg").attr("width", width).attr("height", height).attr("viewBox", `0 0 ${width} ${height}`).style("max-width", "100%");
  const g = svg.append("g").attr("transform", `translate(${cx},${cy})`);
  const arcGen = (a0, a1, r0, r1) => d3.arc().innerRadius(r0).outerRadius(r1).startAngle(a0).endAngle(a1)();
  g.append("path").attr("d", arcGen(-Math.PI / 2, Math.PI / 2, R - 16, R)).attr("fill", accentColor).attr("opacity", 0.2);
  g.append("path").attr("d", arcGen(-Math.PI / 2, Math.PI / 2, R - 16, R)).attr("fill", "none").attr("stroke", "#999").attr("stroke-width", 1);
  [GMIN, GMAX / 2, GMAX].forEach(v => {
    const a = angleFor(v);
    const [tx, ty] = needlePoint(a, R + 18);
    g.append("text").attr("x", tx - cx).attr("y", ty - cy).attr("text-anchor", "middle").attr("font-size", 10).attr("fill", "currentColor").text(formatMin(v));
  });

  const needle = g.append("line").attr("x1", 0).attr("y1", 0).attr("x2", 0).attr("y2", -(R - 24)).attr("stroke", "#333").attr("stroke-width", 3).attr("stroke-linecap", "round");
  g.append("circle").attr("r", 5).attr("fill", "#333");
  const readout = svg.append("text").attr("x", cx).attr("y", cy - R * 0.45).attr("text-anchor", "middle").attr("font-size", 20).attr("font-weight", 700).attr("fill", "currentColor").text(formatMin(0));

  let guessVal = 0;
  function setNeedle(v) {
    const a = angleFor(v);
    const [nx, ny] = needlePoint(a, R - 24);
    needle.attr("x2", nx - cx).attr("y2", ny - cy);
    readout.text(formatMin(v));
  }
  setNeedle(0);

  const slider = html`<input type="range" min="${GMIN}" max="${GMAX}" step="5" value="0" style="width: 100%; max-width: 420px;">`;
  slider.oninput = () => { guessVal = +slider.value; setNeedle(guessVal); };

  const submitBtn = html`<button style="padding:8px 20px;border-radius:4px;border:1px solid #5c9e95;background:#5c9e95;color:white;cursor:pointer;font-size:0.95rem;margin-top: 16px;">Comprobar</button>`;

  card.appendChild(group);
  card.appendChild(question);
  card.appendChild(gaugeHolder);
  card.appendChild(slider);
  card.appendChild(document.createElement("br"));
  card.appendChild(submitBtn);
  container.appendChild(card);

  submitBtn.onclick = () => {
    slider.disabled = true;
    submitBtn.remove();
    const errorAbs = Math.abs(guessVal - data.gapMin);
    const fb = feedback(errorAbs, data.gapMin);
    const fbBox = html`<div style="font-size: 1.05rem; margin: 12px 0;"><span style="font-size:1.4rem;">${fb.emoji}</span> ${fb.text} Adivinaste <strong>${formatMin(guessVal)}</strong>, la brecha real es <strong>${formatMin(data.gapMin)}</strong> (te desviaste por ${formatMin(errorAbs)}).</div>`;

    needle.attr("stroke", "#aaa").attr("stroke-width", 2).attr("stroke-dasharray", "3,2");
    readout.attr("fill", "#aaa");
    const realNeedle = g.append("line").attr("x1", 0).attr("y1", 0).attr("x2", 0).attr("y2", -(R - 24)).attr("stroke", accentColor).attr("stroke-width", 3.5).attr("stroke-linecap", "round");
    g.append("circle").attr("r", 4).attr("fill", accentColor);
    realNeedle.transition().duration(700).attrTween("x2", () => {
      const interp = d3.interpolateNumber(angleFor(guessVal), angleFor(data.gapMin));
      return t => needlePoint(interp(t), R - 24)[0] - cx;
    }).attrTween("y2", () => {
      const interp = d3.interpolateNumber(angleFor(guessVal), angleFor(data.gapMin));
      return t => needlePoint(interp(t), R - 24)[1] - cy;
    });

    card.appendChild(fbBox);
    card.appendChild(contextHoursBarsSvg(data.hoursM, data.hoursH));
    card.appendChild(retryButton(container, buildGaugeDurationDemo));
  };
}
function buildSeesawDurationDemo(container) {

function buildRaceDurationDemo(container) {
  const ageTab = randomAgeTab();
  const data = roundData(ageTab);
  const GMAX = 150;

  const card = html`<div style="max-width: 640px; margin: 0 auto; padding: 24px; border-radius: 10px; background: color-mix(in srgb, currentColor 5%, transparent); text-align:center;"></div>`;
  const group = html`<div style="font-weight: 700; font-size: 1.1rem; margin-bottom: 8px;">${ageTabLabel[ageTab]}</div>`;
  const question = html`<div style="margin-bottom: 16px; max-width: 480px; margin-left:auto; margin-right:auto;">Apuesta cuántos <strong>minutos más</strong> sigue corriendo el cronómetro de las <span style="color:${colorM}; font-weight:700;">mujeres</span> después de que se detiene el de los <span style="color:${colorH}; font-weight:700;">hombres</span> — luego dale a correr y mira quién llega primero.</div>`;

  let guessVal = 0;
  const readout = html`<div style="font-size: 1.1rem; font-weight: 700; margin-bottom: 8px;">${formatMin(0)}</div>`;
  const slider = html`<input type="range" min="0" max="${GMAX}" step="5" value="0" style="width: 100%; max-width: 420px;">`;
  slider.oninput = () => { guessVal = +slider.value; readout.textContent = formatMin(guessVal); };

  function stopwatchRow(label, color) {
    const row = html`<div style="display:flex; align-items:center; gap:12px; margin-bottom: 10px;"></div>`;
    const nameTag = html`<div style="width:70px; text-align:left; font-weight:700; color:${color};">${label}</div>`;
    const track = html`<div style="flex:1; height:16px; background:color-mix(in srgb, currentColor 10%, transparent); border-radius:8px; overflow:hidden;"></div>`;
    const fill = html`<div style="height:100%; width:0%; background:${color}; border-radius:8px;"></div>`;
    track.appendChild(fill);
    const timeLabel = html`<div style="width:110px; text-align:right; font-variant-numeric: tabular-nums; font-weight:700;">0 min</div>`;
    row.appendChild(nameTag);
    row.appendChild(track);
    row.appendChild(timeLabel);
    return { row, fill, timeLabel };
  }

  const raceHolder = html`<div style="max-width: 480px; margin: 0 auto 12px;"></div>`;
  const mRow = stopwatchRow("Mujeres", colorM);
  const hRow = stopwatchRow("Hombres", colorH);
  raceHolder.appendChild(mRow.row);
  raceHolder.appendChild(hRow.row);

  const startBtn = html`<button style="padding:8px 20px;border-radius:4px;border:1px solid #5c9e95;background:#5c9e95;color:white;cursor:pointer;font-size:0.95rem;margin-top:8px;">Iniciar carrera ▶</button>`;

  card.appendChild(group);
  card.appendChild(question);
  card.appendChild(readout);
  card.appendChild(slider);
  card.appendChild(raceHolder);
  card.appendChild(startBtn);
  container.appendChild(card);

  startBtn.onclick = () => {
    slider.disabled = true;
    startBtn.disabled = true;
    startBtn.textContent = "Corriendo...";

    const totalM = data.hoursM * 60, totalH = data.hoursH * 60;
    const maxTotal = Math.max(totalM, totalH);
    const DURATION_MS = 3200;
    const rate = maxTotal / DURATION_MS;
    const t0 = performance.now();
    let hFlagged = false;

    function frame(now) {
      const elapsed = now - t0;
      const simMin = Math.min(elapsed * rate, maxTotal);
      const mVal = Math.min(simMin, totalM);
      const hVal = Math.min(simMin, totalH);
      mRow.fill.style.width = (mVal / maxTotal * 100) + "%";
      hRow.fill.style.width = (hVal / maxTotal * 100) + "%";
      mRow.timeLabel.textContent = formatMin(Math.round(mVal));
      if (hVal >= totalH && !hFlagged) { hFlagged = true; }
      hRow.timeLabel.textContent = formatMin(Math.round(hVal)) + (hFlagged ? " 🏁" : "");
      if (simMin < maxTotal) {
        requestAnimationFrame(frame);
      } else {
        finish();
      }
    }
    requestAnimationFrame(frame);

    function finish() {
      startBtn.remove();
      const errorAbs = Math.abs(guessVal - data.gapMin);
      const fb = feedback(errorAbs, data.gapMin);
      const fbBox = html`<div style="font-size: 1.05rem; margin: 16px 0;"><span style="font-size:1.4rem;">${fb.emoji}</span> ${fb.text} Apostaste <strong>${formatMin(guessVal)}</strong>, la diferencia real es <strong>${formatMin(data.gapMin)}</strong> (te desviaste por ${formatMin(errorAbs)}).</div>`;
      card.appendChild(fbBox);
      card.appendChild(contextHoursBarsSvg(data.hoursM, data.hoursH));
      card.appendChild(retryButton(container, buildRaceDurationDemo));
    }
  };
}

    { id: "thermo", label: "Termómetros", build: buildThermoDurationDemo },
    { id: "race", label: "Carrera de cronómetros", build: buildRaceDurationDemo }
  ];

  let active = "gauge";
  const buttons = templates.map(t => {
    const b = document.createElement("button");
    b.textContent = t.label;
    b.style.cssText = "padding:6px 16px;margin:0 4px 8px;border-radius:4px;border:1px solid #5c9e95;cursor:pointer;font-size:0.9rem;";
    b.onclick = () => { active = t.id; refresh(); };
    tabBar.appendChild(b);
    return { btn: b, id: t.id };
  });

  function updateButtons() {
    buttons.forEach(({ btn, id }) => {
      const on = id === active;
      btn.style.background = on ? "#5c9e95" : "white";
      btn.style.color = on ? "white" : "#333";
    });
  }

  function refresh() {
    updateButtons();
    demoBox.innerHTML = "";
    templates.find(t => t.id === active).build(demoBox);
  }

  refresh();
  return wrapper;
}
