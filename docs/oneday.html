Un día cualquiera

Cómo se distribuyen las actividades de la población a lo largo del día, según la ENUT 2021 (Argentina)

Inspirado en oneday.html de ustimeuse.github.io — pero construido con datos reales de un diario de actividades de 10 minutos, en vez de simulados.

NoteFuente de los datos

Instituto Nacional de Estadística y Censos (INDEC) de Argentina, Encuesta Nacional de Uso del Tiempo 2021 (ENUT 2021) — a diferencia de la ENUT de Chile (que solo registra tiempo total por actividad), la ENUT argentina incluye un diario de actividades en intervalos de 10 minutos a lo largo de las 24 horas del día, lo que permite ver cómo cambia la actividad de la población minuto a minuto. Datos públicos descargados vía la API de datos.gob.ar (sin restricciones de acceso). Las actividades se agrupan en 7 categorías según el clasificador CAUTAL de la propia encuesta. Los porcentajes están ponderados por el factor de expansión de personas (WPER).

oneDayRaw = FileAttachment("data/enut_ar/oneday_by_sex.json").json()
oneDay = oneDayRaw.map(d => ({
  ...d,
  hour: (+d.time.slice(0, 2)) + (+d.time.slice(3, 5)) / 60
}))
grupoOrder = ["Personal", "Trabajo", "Domésticas", "Cuidado", "Ocio", "Voluntarias", "Sin clasificar"]
grupoColors = ["#8ca8c4", "#5c9e95", "#d98c62", "#c7a3c0", "#e0b23e", "#7fa37f", "#c9c9c9"]

1.000 personas, un día

Cada punto representa a 1 de cada 1.000 habitantes. Los puntos se agrupan según la actividad que realizan en cada momento del día — igual que en la visualización original, pero con proporciones reales de la ENUT 2021 en vez de simuladas.

overallRaw = FileAttachment("data/enut_ar/oneday_overall.json").json()
overallByTime = {
  const m = new Map();
  for (const t of new Set(overallRaw.map(d => d.time))) {
    const rows = overallRaw.filter(d => d.time === t);
    m.set(t, Object.fromEntries(rows.map(r => [r.grupo, r.n])));
  }
  return m;
}
overallTimes = [...overallByTime.keys()].sort()
centroids = ({
  "Personal": [500, 300],
  "Trabajo": [200, 150],
  "Domésticas": [800, 150],
  "Ocio": [200, 450],
  "Cuidado": [800, 450],
  "Voluntarias": [500, 80],
  "Sin clasificar": [500, 560]
})
oneDayViz = {
  const width = 1000, height = 600;
  const colorFor = Object.fromEntries(grupoOrder.map((g, i) => [g, grupoColors[i]]));

  function formatTime(t) {
    const [h, m] = t.split(":").map(Number);
    const period = h < 12 ? "am" : "pm";
    let h12 = h % 12;
    if (h12 === 0) h12 = 12;
    return `${h12}:${String(m).padStart(2, "0")}${period}`;
  }

  // build the initial 1000 dots from the first time slot's counts
  function buildDots(t) {
    const counts = overallByTime.get(t);
    const dots = [];
    let idx = 0;
    for (const g of grupoOrder) {
      const n = counts[g] || 0;
      const [cx, cy] = centroids[g];
      for (let i = 0; i < n; i++) {
        dots.push({id: idx++, grupo: g, x: cx + (Math.random() - 0.5) * 20, y: cy + (Math.random() - 0.5) * 20});
      }
    }
    return dots;
  }

  let timeIndex = 0;
  const dots = buildDots(overallTimes[0]);

  const container = html`<div>
    <div style="text-align:center; font-size: 2.2rem; color: var(--bs-secondary-color, #666); font-weight: 300;" class="time-label"></div>
    <div style="text-align:center; margin-bottom: 10px;">
      <button class="btn-play" style="padding: 4px 10px; margin-right: 6px;">⏸ Pausar</button>
      <button class="btn-slow" style="padding: 4px 10px; margin-right: 6px;">Lento</button>
      <button class="btn-fast" style="padding: 4px 10px;">Rápido</button>
    </div>
    <div style="display:flex; gap: 24px; align-items:flex-start; flex-wrap: wrap;">
      <div class="chart-holder"></div>
      <div class="side-panel" style="flex: 1; min-width: 260px;">
        <div class="bar-holder"></div>
      </div>
    </div>
  </div>`;

  const svg = d3.select(container.querySelector(".chart-holder"))
    .append("svg")
    .attr("width", width)
    .attr("height", height)
    .attr("viewBox", [0, 0, width, height])
    .attr("style", "max-width: 100%; height: auto;");

  const pctLabels = {};
  for (const g of grupoOrder) {
    const [cx, cy] = centroids[g];
    svg.append("text")
      .attr("x", cx).attr("y", cy - 46)
      .attr("text-anchor", "middle")
      .attr("font-size", 12)
      .attr("fill", "currentColor")
      .text(g);
    pctLabels[g] = svg.append("text")
      .attr("x", cx).attr("y", cy - 32)
      .attr("text-anchor", "middle")
      .attr("font-size", 11)
      .attr("fill", "currentColor")
      .attr("opacity", 0.7);
  }

  const circles = svg.selectAll("circle")
    .data(dots, d => d.id)
    .join("circle")
    .attr("r", 2.4)
    .attr("fill", d => colorFor[d.grupo])
    .attr("cx", d => d.x)
    .attr("cy", d => d.y)
    .attr("opacity", 0.85);

  const sim = d3.forceSimulation(dots)
    .force("x", d3.forceX(d => centroids[d.grupo][0]).strength(0.1))
    .force("y", d3.forceY(d => centroids[d.grupo][1]).strength(0.1))
    .force("collide", d3.forceCollide(2.6))
    .alphaDecay(0.03)
    .on("tick", () => {
      circles.attr("cx", d => d.x).attr("cy", d => d.y);
    });

  function updateSidePanel(t) {
    const counts = overallByTime.get(t);
    const ranked = grupoOrder.map(g => ({grupo: g, n: counts[g] || 0})).sort((a, b) => b.n - a.n);
    const barHolder = container.querySelector(".bar-holder");
    barHolder.innerHTML = "";
    barHolder.appendChild(Plot.plot({
      width: 320,
      height: 340,
      marginLeft: 130,
      x: {label: "personas de 1.000 →", domain: [0, 1000]},
      y: {domain: ranked.map(d => d.grupo), label: null},
      marks: [
        Plot.barX(ranked, {y: "grupo", x: "n", fill: d => colorFor[d.grupo]}),
        Plot.text(ranked, {y: "grupo", x: "n", dx: 20, text: d => d.n, textAnchor: "start"}),
        Plot.ruleX([0])
      ]
    }));
    for (const g of grupoOrder) {
      pctLabels[g].text(Math.round((counts[g] || 0) / 10) + "%");
    }
    container.querySelector(".time-label").textContent = formatTime(t);
  }

  function advance() {
    timeIndex = (timeIndex + 1) % overallTimes.length;
    const t = overallTimes[timeIndex];
    const target = overallByTime.get(t);

    const currentByGrupo = d3.group(dots, d => d.grupo);
    const pool = [];
    for (const g of grupoOrder) {
      const arr = currentByGrupo.get(g) || [];
      const want = target[g] || 0;
      for (let i = want; i < arr.length; i++) pool.push(arr[i]);
    }
    let poolIdx = 0;
    for (const g of grupoOrder) {
      const arr = currentByGrupo.get(g) || [];
      const want = target[g] || 0;
      for (let i = arr.length; i < want && poolIdx < pool.length; i++, poolIdx++) {
        pool[poolIdx].grupo = g;
      }
    }

    circles.attr("fill", d => colorFor[d.grupo]);
    // d3-force's forceX/forceY accessors are only evaluated once, at
    // initialization -- re-setting the forces after mutating d.grupo makes
    // d3 recompute each node's target, otherwise dots keep the old target
    // forever even after their category (and color) has changed.
    sim.force("x", d3.forceX(d => centroids[d.grupo][0]).strength(0.1));
    sim.force("y", d3.forceY(d => centroids[d.grupo][1]).strength(0.1));
    sim.alpha(0.7).restart();
    updateSidePanel(t);
  }

  let playing = true;
  let speed = 250;
  let timer = setInterval(advance, speed);
  function restartTimer() {
    clearInterval(timer);
    timer = setInterval(advance, speed);
  }

  container.querySelector(".btn-play").onclick = (e) => {
    playing = !playing;
    e.target.textContent = playing ? "⏸ Pausar" : "▶ Reanudar";
    if (playing) restartTimer(); else clearInterval(timer);
  };
  container.querySelector(".btn-slow").onclick = () => { speed = 600; if (playing) restartTimer(); };
  container.querySelector(".btn-fast").onclick = () => { speed = 100; if (playing) restartTimer(); };

  invalidation.then(() => clearInterval(timer));

  updateSidePanel(overallTimes[0]);

  return container;
}
Note

Los 1.000 puntos son una muestra proporcional reconstruida a partir de los porcentajes reales por intervalo de 10 minutos — igual que en la visualización original, no son 1.000 personas reales rastreadas individualmente. En cada paso, solo se mueven los puntos que necesitan cambiar de categoría para que el total vuelva a cuadrar con los datos.

El mismo patrón, desagregado por sexo

// Stack manually (y0/y1 per time+sex, in grupoOrder) instead of relying on
// Plot's automatic stack transform -- more predictable and easier to verify.
stacked = {
  const byKey = d3.group(oneDay, d => `${d.time}|${d.sex}`);
  const rows = [];
  for (const arr of byKey.values()) {
    const base = arr[0];
    let y0 = 0;
    for (const g of grupoOrder) {
      const rec = arr.find(r => r.grupo === g);
      const pct = rec ? rec.pct : 0;
      const y1 = y0 + pct;
      rows.push({time: base.time, hour: base.hour, sex: base.sex, grupo: g, pct, y0, y1});
      y0 = y1;
    }
  }
  return rows;
}
stackedMujeres = stacked.filter(d => d.sex === "Mujeres")
stackedVarones = stacked.filter(d => d.sex === "Varones")
// Two separate plots (not one faceted with fy:"sex") -- Plot.areaY with
// explicit y1/y2 channels renders incorrectly when combined with fy
// faceting in the Plot version bundled here; two plain plots sidestep it.
dayChart = (data, title) => Plot.plot({
  title,
  width: 800,
  height: 260,
  marginLeft: 50,
  x: {label: "Hora del día →", domain: [0, 24], ticks: [0, 3, 6, 9, 12, 15, 18, 21, 24], tickFormat: d => `${d}h`},
  y: {label: "Proporción de la población →", domain: [0, 1], tickFormat: d => `${Math.round(d * 100)}%`},
  color: {legend: true, domain: grupoOrder, range: grupoColors},
  marks: [
    Plot.areaY(data, {x: "hour", y1: "y0", y2: "y1", fill: "grupo", tip: true}),
    Plot.ruleY([0])
  ]
})
dayChart(stackedMujeres, "Mujeres")
dayChart(stackedVarones, "Varones")
Important

El patrón de sueño (categoría “Personal”, que domina la madrugada) es casi idéntico entre mujeres y varones. La diferencia salta a la vista durante el día: los varones muestran un pico de “Trabajo” más pronunciado al mediodía, mientras que “Domésticas” y “Cuidado” ocupan una porción visiblemente mayor del día de las mujeres, sin que su “Trabajo” remunerado disminuya tanto durante esas horas — el patrón visual de la doble jornada.

Cómo se hizo esta página

  1. Se descargaron los archivos enut2021_diario.txt (diario de actividades, ~2 millones de filas: una por persona por intervalo de 10 minutos) y enut2021_base.txt (datos de la persona: sexo, factor de expansión) desde el portal de datos abiertos de Argentina, usando su API CKAN.
  2. Se procesaron con scripts/build_argentina_oneday.R (o su equivalente en Python, scripts/build_argentina_oneday.py): cada actividad se clasificó en una de 7 categorías (clasificador CAUTAL), y se calculó, para cada intervalo de 10 minutos y cada sexo, qué porcentaje ponderado de la población estaba en cada categoría.
  3. El resultado — un archivo pequeño (data/enut_ar/oneday_by_sex.json, ~2.000 filas) — es lo único que salió del procesamiento; no se publica el diario completo.
  4. Esta página lo grafica con Observable Plot como un gráfico de áreas apiladas, directamente en el navegador.
