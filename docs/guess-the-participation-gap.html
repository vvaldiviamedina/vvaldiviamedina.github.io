Adivina la brecha: ¿quién participa más?

Un mini-juego con medidor sobre la brecha de género en tareas domésticas — ENUT Chile 2023

Segunda vuelta del juego, con otra mecánica: esta vez adivinas una diferencia en puntos porcentuales — cuántos puntos más (o menos) de mujeres, comparado con hombres, realizaron cada tarea doméstica en un día típico — moviendo la aguja de un medidor. Ojo: en algunas tareas son los hombres quienes participan más, así que la aguja puede apuntar hacia cualquiera de los dos lados.

NoteFuente de los datos

Instituto Nacional de Estadísticas (INE) de Chile, II Encuesta Nacional sobre Uso del Tiempo (II ENUT) 2023, personas de 15 años y más, población total. Las tasas de participación son las mismas de la rueda en Trabajo pago y doméstico.

function shuffle(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}
activityList = [...new Set(participationRaw.map(d => d.activity_code))].map(code => {
  const rec = participationRaw.find(d => d.activity_code === code && d.age_tab === "Total");
  return { code, label: rec.activity };
})
function getShare(code, sex) {
  const rec = participationRaw.find(d => d.activity_code === code && d.age_tab === "Total" && d.sex === sex);
  return rec ? rec.share : null;
}
function roundData(code) {
  const shareM = getShare(code, "Mujeres");
  const shareH = getShare(code, "Hombres");
  const gap = Math.round((shareM - shareH) * 100);
  return { code, label: activityList.find(a => a.code === code).label, shareM, shareH, gap };
}
function feedback(errorAbs) {
  if (errorAbs <= 3) return { emoji: "🎯", text: "¡Exacto!" };
  if (errorAbs <= 8) return { emoji: "🔥", text: "Caliente." };
  if (errorAbs <= 15) return { emoji: "🌤️", text: "Tibio." };
  return { emoji: "🥶", text: "Frío — bastante lejos." };
}
function fmtPP(v) {
  const r = Math.round(v);
  return (r > 0 ? "+" : "") + r + " pp";
}

    }).attrTween("y2", () => {
      const interp = d3.interpolateNumber(angleFor(guess), angleFor(data.gap));
      return t => needlePointLocal(interp(t), R - 24)[1];

    card.appendChild(table);
    card.appendChild(closing);
    card.appendChild(replayBtn);
    roundBox.appendChild(card);

  return wrapper;

function buildSeesawDemo(container) {
  const code = randomActivityCode();

  const slider = html`<input type="range" min="${GMIN}" max="${GMAX}" step="1" value="0" style="width: 100%; max-width: 420px;">`;
  slider.oninput = () => {
    guessVal = +slider.value;
    setTilt(guessVal);
    readout.textContent = fmtPP(guessVal);
  };

  const submitBtn = html`<button style="padding:8px 20px;border-radius:4px;border:1px solid #5c9e95;background:#5c9e95;color:white;cursor:pointer;font-size:0.95rem;margin-top:16px;">Comprobar</button>`;

  card.appendChild(iconLine);
  card.appendChild(group);
  card.appendChild(question);
  card.appendChild(holder);
  card.appendChild(readout);
  card.appendChild(slider);
  card.appendChild(document.createElement("br"));
  card.appendChild(submitBtn);
  container.appendChild(card);

  submitBtn.onclick = () => {
    slider.disabled = true;
    submitBtn.remove();
    const errorAbs = Math.abs(guessVal - data.gap);
    const fb = feedback(errorAbs);
    const fbBox = html`<div style="font-size: 1.05rem; margin: 12px 0;"><span style="font-size:1.4rem;">${fb.emoji}</span> ${fb.text} Adivinaste <strong>${fmtPP(guessVal)}</strong>, el valor real es <strong>${fmtPP(data.gap)}</strong> (te desviaste por ${errorAbs} pp).</div>`;
    beamGroup.transition().duration(900).attrTween("transform", () => {
      const interp = d3.interpolateNumber(angleFor(guessVal), angleFor(data.gap));
      return t => `rotate(${interp(t)} ${pivotX} ${pivotY})`;
    });
    readout.textContent = fmtPP(data.gap);
    readout.style.color = accentColor;
    const bars = contextBarsSvg(data.shareM, data.shareH);
    card.appendChild(fbBox);
    card.appendChild(bars);
    card.appendChild(retryButton(container, buildSeesawDemo));
  };
}
function buildThermoDemo(container) {

    col.appendChild(valLabel);
    col.appendChild(slider);
    col.appendChild(nameLabel);
    return { col, slider, valLabel };
  }

  const mCol = verticalSlider("Mujeres", colorM);
  const hCol = verticalSlider("Hombres", colorH);
  row.appendChild(mCol.col);
  row.appendChild(hCol.col);

  const gapReadout = html`<div style="font-size: 1.05rem; margin-bottom: 12px;">Brecha implícita: <strong>0 pp</strong></div>`;
  function updateGapReadout() {
    const g = (+mCol.slider.value) - (+hCol.slider.value);
    gapReadout.querySelector("strong").textContent = fmtPP(g);
  }
  mCol.slider.oninput = () => { mCol.valLabel.textContent = mCol.slider.value + "%"; updateGapReadout(); };
  hCol.slider.oninput = () => { hCol.valLabel.textContent = hCol.slider.value + "%"; updateGapReadout(); };

  const submitBtn = html`<button style="padding:8px 20px;border-radius:4px;border:1px solid #5c9e95;background:#5c9e95;color:white;cursor:pointer;font-size:0.95rem;margin-top:8px;">Comprobar</button>`;

  card.appendChild(group);
  card.appendChild(question);
  card.appendChild(row);
  card.appendChild(gapReadout);
  card.appendChild(submitBtn);
  container.appendChild(card);

  submitBtn.onclick = () => {
    mCol.slider.disabled = true;
    hCol.slider.disabled = true;
    submitBtn.remove();
    const guessM = +mCol.slider.value, guessH = +hCol.slider.value;
    const guessGap = guessM - guessH;
    const errorAbs = Math.abs(guessGap - data.gap);
    const fb = feedback(errorAbs);
    const fbBox = html`<div style="font-size: 1.05rem; margin: 12px 0;"><span style="font-size:1.4rem;">${fb.emoji}</span> ${fb.text} Tu combinación implica <strong>${fmtPP(guessGap)}</strong>, la brecha real es <strong>${fmtPP(data.gap)}</strong> (te desviaste por ${errorAbs} pp).</div>`;
    const bars = contextBarsSvg(data.shareM, data.shareH);
    card.appendChild(fbBox);
    card.appendChild(bars);
    card.appendChild(retryButton(container, buildThermoDemo));
  };
}
templateGallery = {
  const wrapper = html`<div id="template-gallery"></div>`;
  const tabBar = html`<div style="text-align:center; margin-bottom: 16px;"></div>`;
  const demoBox = html`<div></div>`;
  wrapper.appendChild(tabBar);
  wrapper.appendChild(demoBox);

  const templates = [
    { id: "line", label: "Línea numérica", build: buildLineDemo },
    { id: "seesaw", label: "Balanza", build: buildSeesawDemo },
    { id: "thermo", label: "Termómetros", build: buildThermoDemo }
  ];

  let active = "line";
  const buttons = templates.map(t => {
    const b = document.createElement("button");
    b.textContent = t.label;
    b.style.cssText = "padding:6px 16px;margin:0 4px 8px;border-radius:4px;border:1px solid #5c9e95;cursor:pointer;font-size:0.9rem;";
    b.onclick = () => { active = t.id; refresh(); };
    tabBar.appendChild(b);
    return { btn: b, id: t.id };
  });

  function updateButtons() {
    buttons.forEach(({ btn, id }) => {
      const on = id === active;
      btn.style.background = on ? "#5c9e95" : "white";
      btn.style.color = on ? "white" : "#333";
    });
  }

  function refresh() {
    updateButtons();
    demoBox.innerHTML = "";
    templates.find(t => t.id === active).build(demoBox);
  }

  refresh();
  return wrapper;
}

Cómo se hizo esta página

  1. Usa los mismos datos que la rueda de Trabajo pago y doméstico: la ENUT 2023 de Chile, procesada con scripts/build_paid_domestic_clock.py (o su equivalente en R).
  2. A diferencia de Adivina la brecha: duración (línea numérica, minutos), el medidor principal de esta página registra la adivinanza con un semicírculo (d3.arc()) y una aguja que gira según el valor — pensado para una diferencia que puede ser negativa (cuando los hombres participan más).
  3. La sección “Prueba otros estilos” son tres bocetos adicionales de la misma mecánica — línea numérica, balanza y termómetros dobles — para comparar qué interacción se siente mejor antes de fijar una sola en el resto del sitio.
