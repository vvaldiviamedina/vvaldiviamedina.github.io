Adivina la brecha: ¿quién participa más?

Un mini-juego con medidor sobre la brecha de género en tareas domésticas — ENUT Chile 2023

Segunda vuelta del juego, con otra mecánica: esta vez adivinas una diferencia en puntos porcentuales — cuántos puntos más (o menos) de mujeres, comparado con hombres, realizaron cada tarea doméstica en un día típico — moviendo la aguja de un medidor. Ojo: en algunas tareas son los hombres quienes participan más, así que la aguja puede apuntar hacia cualquiera de los dos lados.

NoteFuente de los datos

Instituto Nacional de Estadísticas (INE) de Chile, II Encuesta Nacional sobre Uso del Tiempo (II ENUT) 2023, personas de 15 años y más, población total. Las tasas de participación son las mismas de la rueda en Trabajo pago y doméstico.

participationRaw = FileAttachment("data/enut/clock_participation.json").json()
colorH = "#5c9e95"
colorM = "#c7a3c0"
accentColor = "#c26b5a"
activityIcons = ({
  psc: "🍲", lv: "🧹", lrc: "🧺", mrm: "🔧", admnhog: "🧾", comphog: "🛒", cmp: "🐾"
})
function shuffle(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}
activityList = [...new Set(participationRaw.map(d => d.activity_code))].map(code => {
  const rec = participationRaw.find(d => d.activity_code === code && d.age_tab === "Total");
  return { code, label: rec.activity };
})
function getShare(code, sex) {
  const rec = participationRaw.find(d => d.activity_code === code && d.age_tab === "Total" && d.sex === sex);
  return rec ? rec.share : null;
}
function roundData(code) {
  const shareM = getShare(code, "Mujeres");
  const shareH = getShare(code, "Hombres");
  const gap = Math.round((shareM - shareH) * 100);
  return { code, label: activityList.find(a => a.code === code).label, shareM, shareH, gap };
}
function feedback(errorAbs) {
  if (errorAbs <= 3) return { emoji: "🎯", text: "¡Exacto!" };
  if (errorAbs <= 8) return { emoji: "🔥", text: "Caliente." };
  if (errorAbs <= 15) return { emoji: "🌤️", text: "Tibio." };
  return { emoji: "🥶", text: "Frío — bastante lejos." };
}
function fmtPP(v) {
  const r = Math.round(v);
  return (r > 0 ? "+" : "") + r + " pp";
}
gaugeGame = {
  const GMIN = -30, GMAX = 30;
  const R = 140, width = 2 * R + 60, height = R + 90;
  const cx = width / 2, cy = height - 55;

  function angleFor(v) {
    const t = Math.max(GMIN, Math.min(GMAX, v));
    return ((t - GMIN) / (GMAX - GMIN) - 0.5) * Math.PI;
  }
  function needlePoint(angle, len) {
    return [cx + Math.sin(angle) * len, cy - Math.cos(angle) * len];
  }

  const wrapper = html`<div></div>`;
  const roundBox = html`<div></div>`;
  wrapper.appendChild(roundBox);

  let order = shuffle(activityList.map(a => a.code));
  let roundIndex = 0;
  let results = [];

  function drawGaugeBase(svg) {
    const arcGen = (a0, a1, r0, r1) => d3.arc().innerRadius(r0).outerRadius(r1).startAngle(a0).endAngle(a1)();
    const g = svg.append("g").attr("transform", `translate(${cx},${cy})`);
    g.append("path").attr("d", arcGen(-Math.PI / 2, 0, R - 16, R)).attr("fill", colorH).attr("opacity", 0.3);
    g.append("path").attr("d", arcGen(0, Math.PI / 2, R - 16, R)).attr("fill", colorM).attr("opacity", 0.3);
    g.append("path").attr("d", arcGen(-Math.PI / 2, Math.PI / 2, R - 16, R)).attr("fill", "none").attr("stroke", "#999").attr("stroke-width", 1);
    // zero tick
    const [zx1, zy1] = [0, -(R - 16)];
    const [zx2, zy2] = [0, -R];
    g.append("line").attr("x1", zx1).attr("y1", zy1).attr("x2", zx2).attr("y2", zy2).attr("stroke", "#666").attr("stroke-width", 2);
    [GMIN, GMIN / 2, 0, GMAX / 2, GMAX].forEach(v => {
      const a = angleFor(v);
      const [tx, ty] = needlePoint(a, R + 14);
      g.append("text").attr("x", tx).attr("y", ty).attr("text-anchor", "middle").attr("font-size", 10).attr("fill", "currentColor").text(v > 0 ? `+${v}` : `${v}`);
    });
    g.append("text").attr("x", -R - 10).attr("y", 14).attr("text-anchor", "start").attr("font-size", 11).attr("fill", colorH).text("← Hombres");
    g.append("text").attr("x", R + 10).attr("y", 14).attr("text-anchor", "end").attr("font-size", 11).attr("fill", colorM).text("Mujeres →");
    return g;
  }

  function renderRound() {
    roundBox.innerHTML = "";
    const code = order[roundIndex];
    const data = roundData(code);

    const card = html`<div style="max-width: 640px; margin: 0 auto; padding: 24px; border-radius: 10px; background: color-mix(in srgb, currentColor 5%, transparent); text-align:center;"></div>`;

    const progress = html`<div style="font-size: 0.85rem; color: var(--bs-secondary-color, #666); margin-bottom: 4px;">Ronda ${roundIndex + 1} de ${order.length}</div>`;
    const iconLine = html`<div style="font-size: 2.2rem;">${activityIcons[code] || "•"}</div>`;
    const group = html`<div style="font-weight: 700; font-size: 1.1rem; margin-bottom: 8px;">${data.label}</div>`;
    const question = html`<div style="margin-bottom: 12px; max-width: 480px; margin-left:auto; margin-right:auto;">¿Cuántos <strong>puntos porcentuales más</strong> (o menos) de <span style="color:${colorM}; font-weight:700;">mujeres</span>, comparado con <span style="color:${colorH}; font-weight:700;">hombres</span>, realizaron esta tarea en un día típico? Mueve la aguja.</div>`;

    const gaugeHolder = html`<div></div>`;
    const svg = d3.select(gaugeHolder).append("svg").attr("width", width).attr("height", height).attr("viewBox", `0 0 ${width} ${height}`).style("max-width", "100%");
    const g = drawGaugeBase(svg);
    const needle = g.append("line").attr("x1", 0).attr("y1", 0).attr("x2", 0).attr("y2", -(R - 24)).attr("stroke", "#333").attr("stroke-width", 3).attr("stroke-linecap", "round");
    g.append("circle").attr("r", 5).attr("fill", "#333");
    const readout = svg.append("text").attr("x", cx).attr("y", cy - R * 0.45).attr("text-anchor", "middle").attr("font-size", 22).attr("font-weight", 700).attr("fill", "currentColor").text("0 pp");

    let guessVal = 0;
    function setNeedle(v) {
      const a = angleFor(v);
      const [nx, ny] = needlePoint(a, R - 24);
      needle.attr("x2", nx - cx).attr("y2", ny - cy);
      readout.text(fmtPP(v));
    }
    setNeedle(0);

    const slider = html`<input type="range" min="${GMIN}" max="${GMAX}" step="1" value="0" style="width: 100%; max-width: 420px;">`;
    slider.oninput = () => { guessVal = +slider.value; setNeedle(guessVal); };

    const submitBtn = html`<button style="padding:8px 20px;border-radius:4px;border:1px solid #5c9e95;background:#5c9e95;color:white;cursor:pointer;font-size:0.95rem;margin-top: 16px;">Comprobar</button>`;

    card.appendChild(progress);
    card.appendChild(iconLine);
    card.appendChild(group);
    card.appendChild(question);
    card.appendChild(gaugeHolder);
    card.appendChild(slider);
    card.appendChild(document.createElement("br"));
    card.appendChild(submitBtn);
    roundBox.appendChild(card);

    submitBtn.onclick = () => {
      results.push({ code, label: data.label, guess: guessVal, actual: data.gap, shareM: data.shareM, shareH: data.shareH });
      renderReveal(data, guessVal, g, needle, readout);
    };
  }

  function renderReveal(data, guess, g, needle, readout) {
    const card = roundBox.querySelector("div");
    // disable slider/button, add real needle + feedback
    const slider = card.querySelector('input[type="range"]');
    const submitBtn = card.querySelector("button");
    if (slider) slider.disabled = true;
    submitBtn.remove();

    const errorAbs = Math.abs(guess - data.gap);
    const fb = feedback(errorAbs);
    const fbBox = html`<div style="font-size: 1.05rem; margin: 12px 0;"><span style="font-size:1.4rem;">${fb.emoji}</span> ${fb.text} Adivinaste <strong>${fmtPP(guess)}</strong>, el valor real es <strong>${fmtPP(data.gap)}</strong> (te desviaste por ${errorAbs} pp).</div>`;

    // grey out guess needle, animate a new accent needle to the real value
    needle.attr("stroke", "#aaa").attr("stroke-width", 2).attr("stroke-dasharray", "3,2");
    readout.attr("fill", "#aaa");

    function angleFor(v) {
      const GMIN = -30, GMAX = 30;
      return ((v - GMIN) / (GMAX - GMIN) - 0.5) * Math.PI;
    }
    const R = 140;
    function needlePointLocal(angle, len) {
      return [Math.sin(angle) * len, -Math.cos(angle) * len];
    }
    const realNeedle = g.append("line").attr("x1", 0).attr("y1", 0)
      .attr("x2", 0).attr("y2", -(R - 24))
      .attr("stroke", accentColor).attr("stroke-width", 3.5).attr("stroke-linecap", "round");
    g.append("circle").attr("r", 4).attr("fill", accentColor);

    realNeedle.transition().duration(700).attrTween("x2", () => {
      const interp = d3.interpolateNumber(angleFor(guess), angleFor(data.gap));
      return t => needlePointLocal(interp(t), R - 24)[0];
    }).attrTween("y2", () => {
      const interp = d3.interpolateNumber(angleFor(guess), angleFor(data.gap));
      return t => needlePointLocal(interp(t), R - 24)[1];
    });

    // context bars
    const barsHolder = html`<div style="margin-top: 16px; text-align:left; max-width: 420px; margin-left:auto; margin-right:auto;"></div>`;
    const barsSvg = d3.select(barsHolder).append("svg").attr("width", 420).attr("height", 80).attr("viewBox", "0 0 420 80");
    const rows = [
      { label: "Mujeres", share: data.shareM, color: colorM, y: 8 },
      { label: "Hombres", share: data.shareH, color: colorH, y: 44 }
    ];
    rows.forEach(r => {
      barsSvg.append("text").attr("x", 0).attr("y", r.y + 15).attr("font-size", 12).attr("fill", "currentColor").text(r.label);
      barsSvg.append("rect").attr("x", 80).attr("y", r.y).attr("width", 4).attr("height", 22).attr("fill", "#ddd");
      barsSvg.append("rect").attr("x", 80).attr("y", r.y).attr("width", r.share * 260).attr("height", 22).attr("fill", r.color).attr("rx", 3);
      barsSvg.append("text").attr("x", 86 + r.share * 260).attr("y", r.y + 15).attr("font-size", 12).attr("fill", "currentColor").text(`${Math.round(r.share * 100)}%`);
    });

    const nextBtn = html`<button style="padding:8px 20px;border-radius:4px;border:1px solid #5c9e95;background:#5c9e95;color:white;cursor:pointer;font-size:0.95rem;margin-top:16px;">${roundIndex + 1 < order.length ? "Siguiente ronda →" : "Ver resultado final"}</button>`;
    nextBtn.onclick = () => {
      if (roundIndex + 1 < order.length) {
        roundIndex += 1;
        renderRound();
      } else {
        renderSummary();
      }
    };

    card.appendChild(fbBox);
    card.appendChild(barsHolder);
    card.appendChild(nextBtn);
  }

  function renderSummary() {
    roundBox.innerHTML = "";
    const card = html`<div style="max-width: 640px; margin: 0 auto; padding: 24px; border-radius: 10px; background: color-mix(in srgb, currentColor 5%, transparent);"></div>`;

    const title = html`<div style="font-weight: 700; font-size: 1.2rem; margin-bottom: 16px;">Resultado final</div>`;

    const table = html`<table style="width:100%; border-collapse: collapse; margin-bottom: 16px;"></table>`;
    const thead = html`<thead><tr>
      <th style="text-align:left; padding:6px; border-bottom: 1px solid #999;">Tarea</th>
      <th style="text-align:right; padding:6px; border-bottom: 1px solid #999;">Tu respuesta</th>
      <th style="text-align:right; padding:6px; border-bottom: 1px solid #999;">Real</th>
      <th style="text-align:right; padding:6px; border-bottom: 1px solid #999;">Diferencia</th>
    </tr></thead>`;
    table.appendChild(thead);
    const tbody = html`<tbody></tbody>`;
    let totalError = 0;
    results.forEach(r => {
      const err = Math.abs(r.guess - r.actual);
      totalError += err;
      const row = html`<tr>
        <td style="padding:6px; border-bottom: 1px solid #eee;">${activityIcons[r.code] || ""} ${r.label}</td>
        <td style="text-align:right; padding:6px; border-bottom: 1px solid #eee;">${fmtPP(r.guess)}</td>
        <td style="text-align:right; padding:6px; border-bottom: 1px solid #eee;">${fmtPP(r.actual)}</td>
        <td style="text-align:right; padding:6px; border-bottom: 1px solid #eee;">${err} pp</td>
      </tr>`;
      tbody.appendChild(row);
    });
    table.appendChild(tbody);

    const avgError = Math.round((totalError / results.length) * 10) / 10;
    const closing = html`<div style="margin-bottom: 20px;">Tu error promedio fue de <strong>${avgError} pp</strong> por ronda. ${avgError <= 5 ? "Excelente calibración." : avgError <= 12 ? "Una intuición razonable, aunque algunas tareas sorprenden." : "La brecha real varía más de lo que la mayoría intuye, tarea por tarea."}</div>`;

    const replayBtn = html`<button style="padding:8px 20px;border-radius:4px;border:1px solid #5c9e95;background:#5c9e95;color:white;cursor:pointer;font-size:0.95rem;">Jugar de nuevo</button>`;
    replayBtn.onclick = () => {
      order = shuffle(activityList.map(a => a.code));
      roundIndex = 0;
      results = [];
      renderRound();
    };

    card.appendChild(title);
    card.appendChild(table);
    card.appendChild(closing);
    card.appendChild(replayBtn);
    roundBox.appendChild(card);
  }

  renderRound();
  return wrapper;
}
Important

“Mantenimiento del hogar” es la única tarea de esta lista donde los hombres participan más que las mujeres — todas las demás muestran brechas a favor de una mayor participación femenina, aunque de magnitudes muy distintas. Revisa la rueda interactiva en Trabajo pago y doméstico para el detalle completo.

Cómo se hizo esta página

  1. Usa los mismos datos que la rueda de Trabajo pago y doméstico: la ENUT 2023 de Chile, procesada con scripts/build_paid_domestic_clock.py (o su equivalente en R).
  2. A diferencia de Adivina la brecha: duración (línea numérica, minutos), aquí la adivinanza se registra con un medidor semicircular (d3.arc()) y una aguja que gira según el valor — pensado para una diferencia que puede ser negativa (cuando los hombres participan más).
